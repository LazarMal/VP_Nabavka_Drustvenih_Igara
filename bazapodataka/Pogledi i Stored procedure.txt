USE `reklamacije_drustvenih_igara_vp_2026`;

DROP VIEW IF EXISTS `svipodacioidrutvenimigramasaslikom`;
DROP VIEW IF EXISTS `svipodacioidrutvenimigrama`;

CREATE VIEW `svipodacioidrutvenimigrama` AS
SELECT 
    `drustvena_igra`.`SifraIgre`,
    `drustvena_igra`.`Naziv`,
    `drustvena_igra`.`Proizvodjac`,
    `kategorija_igre`.`Naziv` AS `NazivKategorije`
FROM `drustvena_igra`
INNER JOIN `kategorija_igre`
ON `drustvena_igra`.`OznakaKategorije` = `kategorija_igre`.`Oznaka`;

CREATE VIEW `svipodacioidrutvenimigramasaslikom` AS
SELECT 
    `drustvena_igra`.`SifraIgre`,
    `drustvena_igra`.`Naziv`,
    `drustvena_igra`.`Proizvodjac`,
    `kategorija_igre`.`Naziv` AS `NazivKategorije`,
    `drustvena_igra`.`NazivFajlaSlike`
FROM `drustvena_igra`
INNER JOIN `kategorija_igre`
ON `drustvena_igra`.`OznakaKategorije` = `kategorija_igre`.`Oznaka`;

DROP PROCEDURE IF EXISTS `DodajDrustvenuIgru`;

DELIMITER $$

CREATE PROCEDURE `DodajDrustvenuIgru`
(
    IN SifraIgreParametar VARCHAR(13),
    IN NazivParametar VARCHAR(100),
    IN ProizvodjacParametar VARCHAR(100),
    IN OznakaKategorijeParametar CHAR(2),
    IN NazivFajlaSlikeParametar VARCHAR(100)
)
BEGIN
    DECLARE EXIT HANDLER FOR SQLEXCEPTION
    BEGIN
        ROLLBACK;
        RESIGNAL;
    END;

    START TRANSACTION;

    INSERT INTO `drustvena_igra`
    (`SifraIgre`, `Naziv`, `Proizvodjac`, `OznakaKategorije`, `NazivFajlaSlike`)
    VALUES
    (SifraIgreParametar, NazivParametar, ProizvodjacParametar, OznakaKategorijeParametar, NazivFajlaSlikeParametar);

    UPDATE `kategorija_igre`
    SET `UkupanBrojIgara` = `UkupanBrojIgara` + 1
    WHERE `Oznaka` = OznakaKategorijeParametar;

    COMMIT;
END $$

DELIMITER ;
