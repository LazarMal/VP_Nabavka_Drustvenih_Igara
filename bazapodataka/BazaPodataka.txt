CREATE DATABASE IF NOT EXISTS `nabavka_drustvenih_igara_vp_2026`
CHARACTER SET utf8mb4
COLLATE utf8mb4_unicode_ci;

USE `nabavka_drustvenih_igara_vp_2026`;

CREATE TABLE `kategorija_igre`
(
   `Oznaka` CHAR(2) NOT NULL PRIMARY KEY,
   `Naziv` VARCHAR(50) NOT NULL,
   `UkupanBrojIgara` INT NOT NULL DEFAULT 0
);

CREATE TABLE `drustvena_igra`
(
   `SifraIgre` VARCHAR(13) NOT NULL PRIMARY KEY,
   `Naziv` VARCHAR(100) NOT NULL,
   `Proizvodjac` VARCHAR(100) NOT NULL,
   `OznakaKategorije` CHAR(2) NOT NULL,
   `NazivFajlaSlike` VARCHAR(100) NULL,
   `Cena` DECIMAL(10,2) NULL,
   CONSTRAINT `FK_IGRA_KATEGORIJA`
   FOREIGN KEY (`OznakaKategorije`) REFERENCES `kategorija_igre`(`Oznaka`)
   ON DELETE RESTRICT
   ON UPDATE CASCADE
);

CREATE TABLE `KORISNIK`
(
   `IDKORISNIKA` INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
   `PREZIME` VARCHAR(50) NOT NULL,
   `IME` VARCHAR(40) NOT NULL,
   `EMAIL` VARCHAR(60) NOT NULL,
   `KORISNICKOIME` VARCHAR(30) NOT NULL,
   `SIFRA` VARCHAR(30) NOT NULL,
   `URLSLike` VARCHAR(250) NULL,
   `statusucesca` VARCHAR(30) NOT NULL
);

CREATE TABLE `nabavka`
(
   `IDNabavke` INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
   `BrojNaloga` VARCHAR(50) NOT NULL,
   `DatumNabavke` DATE NOT NULL,
   `Dobavljac` VARCHAR(100) NOT NULL,
   `Napomena` VARCHAR(255) NULL,
   `NalogEvidentirao` VARCHAR(100) NOT NULL,
   CONSTRAINT `UQ_NABAVKA_BROJNALOGA` UNIQUE (`BrojNaloga`)
);

CREATE TABLE `stavka_nabavke`
(
   `IDStavke` INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
   `IDNabavke` INT NOT NULL,
   `SifraIgre` VARCHAR(13) NOT NULL,
   `Kolicina` INT NOT NULL,
   `Cena` DECIMAL(10,2) NOT NULL,
   CONSTRAINT `FK_STAVKA_NABAVKA`
   FOREIGN KEY (`IDNabavke`) REFERENCES `nabavka`(`IDNabavke`)
   ON DELETE CASCADE
   ON UPDATE CASCADE,
   CONSTRAINT `FK_STAVKA_IGRA`
   FOREIGN KEY (`SifraIgre`) REFERENCES `drustvena_igra`(`SifraIgre`)
   ON DELETE RESTRICT
   ON UPDATE CASCADE
);
